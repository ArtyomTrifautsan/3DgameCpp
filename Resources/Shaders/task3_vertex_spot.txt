#version 330 core

layout(location = 0) in vec3 aPos;
layout(location = 0) in vec3 aNormal;

out vec3 vertexColor;

uniform mat4 model_matrix;
uniform mat4 view_projection_matrix;
uniform mat3 normal_matrix;

uniform float uMorphFactor; // 0.0 = куб, 1.0 = сфера

uniform vec3 camera_pos;
uniform vec3 light_pos;
uniform vec3 light_dir;
uniform vec3 light_color;

uniform vec3 ambient_color;
uniform float ambient_factor;
uniform vec3 diffuse_color;
uniform float diffuse_factor;
uniform vec3 specular_color;
uniform float specular_factor;
uniform float shininess;

uniform vec3 attenuation;

uniform float cutOff;       // Внутренний угол (косинус)
uniform float outerCutOff;  // Внешний угол (косинус) для мягких краев

void main()
{
    //      === Морфирование ===
    vec3 morphed_pos = mix(aPos, normalize(aPos), uMorphFactor);
    vec3 morphed_normal = mix(aNormal, normalize(aNormal), uMorphFactor);

    gl_Position = view_projection_matrix * model_matrix * vec4(morphed_pos, 1.0);

    // Префикс w - world
    vec3 w_vertex_pos = vec3(model_matrix * vec4(morphed_pos, 1.0));
    vec3 w_normal = normalize(normal_matrix * morphed_normal);


    // Здесь light.direction - это light_dir, lightDir - это lightDir
    //      === Расчет света ===
    // Ambient
    vec3 ambient  = light_color  * ambient_color * ambient_factor;

    // Diffuse
    vec3 lightDir = normalize(light_pos - w_vertex_pos);  
    float diff = max(dot(w_normal, lightDir), 0.0);
    vec3 diffuse  = light_color  * diff * diffuse_color * diffuse_factor;
    
    // Specular
    vec3 view_dir = normalize(camera_pos - w_vertex_pos);
    vec3 reflectDir = reflect(-lightDir, w_normal);
    float specular_value = pow(max(dot(view_dir, reflectDir), 0.0), shininess);
    vec3 specular = light_color * specular_value * specular_value * specular_color;

    // Затухание
    float distance = length(light_pos - w_vertex_pos);
    float attenuation_value = 1.0 / (attenuation.x + attenuation.y * distance + attenuation.z * (distance * distance));

    // Угол прожектора
    float theta     = dot(lightDir, normalize(-light_dir));
    float epsilon   = cutOff - outerCutOff;
    float intensity = clamp((theta - outerCutOff) / epsilon, 0.0, 1.0);
    
    ambient  *= attenuation_value;
    diffuse  *= attenuation_value * intensity;
    specular *= attenuation_value * intensity;
    
    vertexColor = ambient + diffuse + specular;
}
